(function () {
  class SmokeTest extends HTMLElement {
    connectedCallback() {
      this.style.display = 'block';
      if (!this.style.minHeight) this.style.minHeight = '220px';
      const root = this.attachShadow({ mode: 'open' });
      const style = document.createElement('style');
      style.textContent = `
        :host { position: relative; display: block; }
        .box { position:absolute; inset:0; display:flex; align-items:center; justify-content:center;
               background:#101318; color:#27e46b; font:14px system-ui; border:1px dashed #27e46b55; }
      `;
      const box = document.createElement('div');
      box.className = 'box';
      box.textContent = 'Velo file wired ✅ ' + new Date().toLocaleTimeString();
      root.appendChild(style);
      root.appendChild(box);
      console.log('[smoke-test] connected');
    }
  }
  try { customElements.define('smoke-test', SmokeTest); } catch {}
})();
